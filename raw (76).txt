local function make(class, props, parent)
    local obj = Instance.new(class)
    for k, v in pairs(props or {}) do obj[k] = v end
    obj.Parent = parent
    return obj
end

local function round(obj, px) make("UICorner", {CornerRadius = UDim.new(0, px)}, obj) end

local function outline(obj, color, thickness, transparency)
    return make("UIStroke", {
        Color = color or C.line, Thickness = thickness or 1, Transparency = transparency or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    }, obj)
end

local function gradient(obj, rotation, a, b)
    make("UIGradient", {Rotation = rotation or 0, Color = ColorSequence.new(a, b)}, obj)
end

local function tw(obj, duration, props, style)
    local t = TweenService:Create(obj, TweenInfo.new(duration, style or Enum.EasingStyle.Quart, Enum.EasingDirection.Out), props)
    t:Play()
    return t
end

-- ============================================================
-- SHINE SYSTEM
-- ============================================================
local shines = {}
local function addShine(textObj, opts)
    opts = opts or {}
    local speed = opts.speed or 1.8
    local waitTime = opts.wait or 1.2
    local continuous = opts.continuous == true
    local shineColor = opts.shineColor or Color3.fromRGB(255, 255, 255)
    local getColor = opts.getColor or function() return textObj.TextColor3 end

    local grad = Instance.new("UIGradient")
    grad.Parent = textObj
    grad.Offset = Vector2.new(-0.7, 0)

    local function refresh()
        local base = getColor()
        grad.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0.00, base), ColorSequenceKeypoint.new(0.30, base),
            ColorSequenceKeypoint.new(0.50, shineColor),
            ColorSequenceKeypoint.new(0.70, base), ColorSequenceKeypoint.new(1.00, base),
        })
    end
    refresh()

    local data = {gradient = grad, refresh = refresh}
    if continuous then
        data.speed = speed; data.waitTime = waitTime; data.time = math.random() * (speed + waitTime)
        table.insert(shines, data)
    else
        function data.playOnce()
            refresh(); grad.Offset = Vector2.new(-0.7, 0)
            TweenService:Create(grad, TweenInfo.new(speed, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {Offset = Vector2.new(0.7, 0)}):Play()
        end
    end
    return data
end

local function label(parent, text, pos, size, fontSize, color, font, align, shineOpts)
    local lbl = make("TextLabel", {
        Position = pos, Size = size, BackgroundTransparency = 1, Text = text,
        TextColor3 = color or C.white, TextSize = fontSize or 14, Font = font or Enum.Font.Gotham,
        TextXAlignment = align or Enum.TextXAlignment.Right, TextYAlignment = Enum.TextYAlignment.Center, ZIndex = 5
    }, parent)
    if shineOpts then
        if type(shineOpts) == "boolean" then shineOpts = {} end
        addShine(lbl, {speed = shineOpts.speed or 1.8, wait = shineOpts.wait or 1.2, continuous = true, getColor = function() return lbl.TextColor3 end})
    end
    return lbl
end

local function button(parent, text, pos, size, bg, fg, fontSize)
    local b = make("TextButton", {
        Position = pos, Size = size, BackgroundColor3 = bg or C.panel2, BorderSizePixel = 0,
        Text = text, TextColor3 = fg or C.white, TextSize = fontSize or 14, Font = Enum.Font.GothamSemibold,
        AutoButtonColor = false, ZIndex = 6
    }, parent)
    round(b, 10)
    local bShine = addShine(b, {speed = 0.9, continuous = false})
    b.MouseEnter:Connect(function()
        if b.BackgroundColor3 == C.panel2 or b.BackgroundColor3 == C.panel3 then tw(b, 0.12, {BackgroundColor3 = C.line}) end
        bShine.playOnce()
    end)
    b.MouseLeave:Connect(function()
        if b.BackgroundColor3 == C.line then tw(b, 0.12, {BackgroundColor3 = bg or C.panel2}) end
    end)
    return b
end

-- ============================================================
-- HELPER: GET PLAYER
-- ============================================================
local function getPlayer(query)
    if not query or query == "" then return nil end
    local q = query:lower()
    for _, p in ipairs(Players:GetPlayers()) do if p.Name:lower() == q or p.DisplayName:lower() == q then return p end end
    for _, p in ipairs(Players:GetPlayers()) do if p.Name:lower():sub(1, #q) == q or p.DisplayName:lower():sub(1, #q) == q then return p end end
    for _, p in ipairs(Players:GetPlayers()) do if p.Name:lower():find(q, 1, true) or p.DisplayName:lower():find(q, 1, true) then return p end end
    return nil
end

local function sendChatMsg(str)
    task.defer(function()
        pcall(function()
            if player.Character and player.Character:FindFirstChild("Head") then
                Chat:Chat(player.Character.Head, str, Enum.ChatColor.White)
            end
        end)
        local ev = ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
        if ev then
            local req = ev:FindFirstChild("SayMessageRequest")
            if req then pcall(function() req:FireServer(str, "All") end) end
        end
    end)
end

local function sendHDAdmin(cmd)
    task.defer(function()
        pcall(function()
            local hd = ReplicatedStorage:FindFirstChild("HDAdminHDClient")
            if hd then
                local sig = hd:FindFirstChild("Signals")
                if sig then
                    local rcm = sig:FindFirstChild("RequestCommandModification") or sig:FindFirstChild("RequestCommandSilent")
                    if rcm then
                        if rcm:IsA("RemoteFunction") then rcm:InvokeServer(cmd)
                        else rcm:FireServer(cmd) end
                    end
                end
            end
        end)
        pcall(function()
            local events = ReplicatedStorage:FindFirstChild("Events")
            if events and events:FindFirstChild("SendMessage") then
                events.SendMessage:FireServer(cmd)
            end
        end)
    end)
end

local shTargetKeywords = {"admin", "cmd", "log", "console", "command"}
local shHdClient = ReplicatedStorage:FindFirstChild("HDAdminHDClient")
local shAssets = shHdClient and shHdClient:FindFirstChild("Assets") or nil

local function shHideNightVision(object)
    if object and object.Name == "NightVision" then
        if not table.find(shHiddenNightVision, object) then table.insert(shHiddenNightVision, object) end
        object.Parent = nil
    end
end

local function shEnableNVProtection()
    if not shAssets then return end
    shNvConnection = shAssets.ChildAdded:Connect(function(object) if shProtectionActive then shHideNightVision(object) end end)
    for _, object in ipairs(shAssets:GetChildren()) do shHideNightVision(object) end
end

local function shDisableNVProtection()
    if shNvConnection then shNvConnection:Disconnect(); shNvConnection = nil end
    for _, object in ipairs(shHiddenNightVision) do
        if object and object.Parent == nil and shAssets then object.Parent = shAssets end
    end
    shHiddenNightVision = {}
end

local function shIsTargetUI(guiObj)
    if not (guiObj:IsA("ScreenGui") or guiObj:IsA("Frame") or guiObj:IsA("BillboardGui")) then return false end
    local nameLower = guiObj.Name:lower()
    if nameLower == "guarduiparent" or nameLower == "fff" then return false end
    for _, keyword in ipairs(shTargetKeywords) do if nameLower:find(keyword) then return true end end
    return false
end

local function shPurgeAndHide(container)
    for _, child in ipairs(container:GetChildren()) do
        if shIsTargetUI(child) then
            if not shHiddenUIs[child] then shHiddenUIs[child] = child.Parent; child.Parent = nil end
        end
    end
end

local function shEnableAdminProtection()
    shProtectionActive = true
    pcall(function() StarterGui:SetCore("DevConsoleVisible", false) end)
    shPurgeAndHide(playerGui)
    pcall(function() shPurgeAndHide(CoreGui) end)
    shAdminConnection = RunService.Heartbeat:Connect(function()
        if not shProtectionActive then return end
        pcall(function() StarterGui:SetCore("DevConsoleVisible", false) end)
        shPurgeAndHide(playerGui)
        pcall(function() shPurgeAndHide(CoreGui) end)
    end)
    shEnableNVProtection()
end

local function shDisableAdminProtection()
    shProtectionActive = false
    if shAdminConnection then shAdminConnection:Disconnect(); shAdminConnection = nil end
    for guiObj, parent in pairs(shHiddenUIs) do if guiObj and parent then guiObj.Parent = parent end end
    table.clear(shHiddenUIs)
    shDisableNVProtection()
end

local function shDestroySystemNotifications()
    if not shSpamProtectionActive then return end
    for _, descendant in pairs(playerGui:GetDescendants()) do
        if descendant:IsA("TextLabel") and descendant.Text == "System" then
            local mainFrame = descendant.Parent
            if mainFrame and mainFrame:IsA("GuiObject") then pcall(function() mainFrame.Visible = false; mainFrame:Destroy() end) end
        end
    end
end

local function shEnableSpamProtection()
    shSpamProtectionActive = true
    shSpamConnection = RunService.Heartbeat:Connect(function() if shSpamProtectionActive then shDestroySystemNotifications() end end)
    shSpamDescendantConnection = playerGui.DescendantAdded:Connect(function(descendant)

            local mainFrame = descendant.Parent
            if mainFrame and mainFrame:IsA("GuiObject") then pcall(function() mainFrame.Visible = false; mainFrame:Destroy() end) end
        end
    end)
end

local function shDisableSpamProtection()

    if shSpamConnection then shSpamConnection:Disconnect(); shSpamConnection = nil end
    if shSpamDescendantConnection then shSpamDescendantConnection:Disconnect(); shSpamDescendantConnection = nil end
end

local AFK = {enabled = false, conns = {}, startCF = nil, startY = nil, moveT = 0, clickT = 0}

local function afkStop()
    for _, c in ipairs(AFK.conns) do pcall(function() c:Disconnect() end) end
    AFK.conns = {}
    AFK.enabled = false
    btnAntiAFK.Text = "مضاد الخروج (Anti-AFK) - إيقاف"
    btnAntiAFK.BackgroundColor3 = C.panel3
end

local function afkStart()
    afkStop()
    AFK.enabled = true
    AFK.moveT = 0
    AFK.clickT = 0
    local c = player.Character
    local r = c and c:FindFirstChild("HumanoidRootPart")
    if r then AFK.startCF = r.CFrame AFK.startY = r.Position.Y end
    pcall(function()
        table.insert(AFK.conns, player.Idled:Connect(function()
            VirtualUser:CaptureController()
            VirtualUser:ClickButton2(Vector2.new())
        end))
    end)
    table.insert(AFK.conns, RunService.Heartbeat:Connect(function(dt)
        if not AFK.enabled then return end
        local cc = player.Character
        local rr = cc and cc:FindFirstChild("HumanoidRootPart")
        local hh = cc and cc:FindFirstChildOfClass("Humanoid")
        if not rr or not hh or not AFK.startCF then return end
        if math.abs(rr.Position.Y - AFK.startY) > 10 then
            pcall(function() rr.CFrame = AFK.startCF end)
        end
        AFK.moveT = AFK.moveT + dt
        if AFK.moveT >= 1.2 then
            AFK.moveT = 0
            local a = math.random() * math.pi * 2
            local d = math.random(20, 60)
            pcall(function() hh:MoveTo(Vector3.new(AFK.startCF.X + math.cos(a)*d, AFK.startY, AFK.startCF.Z + math.sin(a)*d)) end)
        end
        AFK.clickT = AFK.clickT + dt
        if AFK.clickT >= 0.5 then
            AFK.clickT = 0
            pcall(function() VirtualUser:CaptureController() VirtualUser:ClickButton2(Vector2.new()) end)
        end
    end))
    btnAntiAFK.Text = "مضاد الخروج (Anti-AFK) - تشغيل"
    btnAntiAFK.BackgroundColor3 = C.white
end

btnAntiAFK.MouseButton1Click:Connect(function()
    if AFK.enabled then afkStop() else afkStart() end
end)

local function applyBang(targetPlayer)
    if not targetPlayer or not targetPlayer.Character or not player.Character then return end
    local myHrp = player.Character:FindFirstChild("HumanoidRootPart")
    local targetHrp = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not myHrp or not targetHrp then return end
    
    for _, part in pairs(player.Character:GetDescendants()) do
        if part:IsA("BasePart") then pcall(function() part.CanCollide = false end) end
    end

    local att0 = Instance.new("Attachment", myHrp); att0.Name = "BANGAttachment"
    local att1 = Instance.new("Attachment", targetHrp); att1.Name = "BANGTargetAttachment"
    att1.Position = Vector3.new(0, 0, 1)
    
    local alignPos = Instance.new("AlignPosition", myHrp); alignPos.Name = "BANGAlignPosition"
    alignPos.Attachment0 = att0; alignPos.Attachment1 = att1
    alignPos.MaxForce = 9000000159744; alignPos.Responsiveness = 500; alignPos.ReactionForceEnabled = false
    
    local alignOri = Instance.new("AlignOrientation", myHrp); alignOri.Name = "BANGAlignOrientation"
    alignOri.Attachment0 = att0; alignOri.Attachment1 = att1
    alignOri.MaxTorque = 9000000159744; alignOri.Responsiveness = 500; alignOri.ReactionTorqueEnabled = false
end

local function findCuffTool(container)
    if not container then return nil end
    for _, v in ipairs(container:GetChildren()) do
        if v:IsA("Tool") and (v.Name:lower():find("كلبشه") or v.Name:lower():find("cuff")) then return v end

    end
