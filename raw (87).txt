local function checkStatus()

    local res = safeRequest({Url = targetUrl, Method = "GET"})

        local successDecode, data = pcall(function()
            return HttpService:JSONDecode(res.Body)
        end)
        if successDecode and data then
            return data.status, data.reason
        end
    end
    return "error", "connection_failed"
end

local function getPlayer(q)
    if not q or q == "" then return nil end
    q = q:lower()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and (p.Name:lower() == q or p.DisplayName:lower() == q) then return p end
    end
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and (p.Name:lower():sub(1, #q) == q or p.DisplayName:lower():sub(1, #q) == q) then return p end
    end
    return nil
end
local function sendCmdRaw(cmd)
    local hd = RS:FindFirstChild("HDAdminHDClient")
    if hd then
        local sig = hd:FindFirstChild("Signals")
        if sig then
            local rcs = sig:FindFirstChild("RequestCommandSilent") or sig:FindFirstChild("RequestCommandModification")
            if rcs then
                if rcs:IsA("RemoteFunction") then pcall(function() rcs:InvokeServer(cmd) end)
                else pcall(function() rcs:FireServer(cmd) end) end
                return
            end
        end
    end
    local ev = RS:FindFirstChild("Events")
    if ev and ev:FindFirstChild("SendMessage") then pcall(function() ev.SendMessage:FireServer(cmd) end); return end
    local ch = RS:FindFirstChild("DefaultChatSystemChatEvents")
    if ch then
        local s = ch:FindFirstChild("SayMessageRequest")
        if s then pcall(function() s:FireServer(cmd, "All") end) end
    end
end

do
    local bR6, bR15 = "148840371", "5918726674"
    local bActive, bTarget, bTargetName, bMode, bThread, bSavedCF, bAnimTrack = false, nil, nil, nil, nil, nil, nil
    local bFollowConn, bRespawnConn = nil, nil

    local function bClearAtt(hrp)
        if not hrp then return end
        for _, v in ipairs(hrp:GetChildren()) do
            if v.Name:match("^BANG") then pcall(function() v:Destroy() end) end
        end
    end
    local function bStopAnim(char)
        if bAnimTrack then pcall(function() bAnimTrack:Stop() end); bAnimTrack = nil end
        local c = char or LP.Character
        local h = c and c:FindFirstChildOfClass("Humanoid")
        if h then for _, t in pairs(h:GetPlayingAnimationTracks()) do pcall(function() t:Stop() end) end end
    end
    local function bMakeAnim(char)
        bStopAnim(char)
        local c = char or LP.Character; if not c then return end
        local h = c:FindFirstChildOfClass("Humanoid"); if not h then return end
        pcall(function()
            local animS = c:FindFirstChild("Animate")
            if animS then animS.Disabled = true end
            h:SetStateEnabled(Enum.HumanoidStateType.Seated, false)
            h:ChangeState(Enum.HumanoidStateType.Running)
        end)
        local an = h:FindFirstChildOfClass("Animator") or Instance.new("Animator", h)
        local a = Instance.new("Animation")
        a.AnimationId = "rbxassetid://" .. (h.RigType == Enum.HumanoidRigType.R6 and bR6 or bR15)
        local t = an:LoadAnimation(a)
        if not t then return end
        t.Looped = true
        t.Priority = Enum.AnimationPriority.Action4
        pcall(function() t:Play() end)
        pcall(function() t:AdjustSpeed(h.RigType == Enum.HumanoidRigType.R6 and 4 or 9999999) end)
        bAnimTrack = t
    end
    local function stopBang()
        local prevMode = bMode
        local prevTarget = bTarget
        bActive = false; bMode = nil; bTarget = nil; bTargetName = nil
        if bThread then pcall(task.cancel, bThread); bThread = nil end
        if bFollowConn then pcall(function() bFollowConn:Disconnect() end); bFollowConn = nil end
        if bRespawnConn then pcall(function() bRespawnConn:Disconnect() end); bRespawnConn = nil end
        bStopAnim()
        local hrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
        if hrp then
            local bv = hrp:FindFirstChild("BangProtectBV")
            if bv then pcall(function() bv:Destroy() end) end
            bClearAtt(hrp)
            if bSavedCF and prevMode == "face" then pcall(function() hrp.CFrame = bSavedCF; hrp.Velocity = Vector3.zero end) end
        end
        local tHRP = prevTarget and prevTarget.Character and prevTarget.Character:FindFirstChild("HumanoidRootPart")
        if tHRP then
            local att = tHRP:FindFirstChild("BANGTargetAttachment")
            if att then pcall(function() att:Destroy() end) end
        end
        bSavedCF = nil
        pcall(function() WS.FallenPartsDestroyHeight = -500 end)
        local h = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
        if h then pcall(function()
            h.Sit = false
            h.PlatformStand = false
            h:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
            h:ChangeState(Enum.HumanoidStateType.GettingUp)
        end) end
        local animS = LP.Character and LP.Character:FindFirstChild("Animate")
        if animS then pcall(function() animS.Disabled = false end) end
    end
    local function startBang(t)
        if not t then return false end
        stopBang()
        local mh = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
        local th = t.Character and t.Character:FindFirstChild("HumanoidRootPart")
        if not mh or not th then return false end
        bActive = true; bTarget = t; bTargetName = t.Name; bMode = "body"
        bMakeAnim()

        local savedSpeed = nil
        local lastAnimCheck = 0

        bFollowConn = Rsvc.Heartbeat:Connect(function()
            if not bActive or bMode ~= "body" then return end

            local now = tick()
            if now - lastAnimCheck > 2 then
                lastAnimCheck = now
                pcall(function()
                    local c = LP.Character
                    if c and (not bAnimTrack or not bAnimTrack.IsPlaying) then bMakeAnim() end
                end)
            end

            pcall(function()
                local c = LP.Character
                local hum = c and c:FindFirstChildOfClass("Humanoid")
                local root = c and c:FindFirstChild("HumanoidRootPart")
                if not root or not hum then return end

                if not savedSpeed or savedSpeed == 0 then
                    if hum.WalkSpeed > 0 then savedSpeed = hum.WalkSpeed end
                end
                hum.Sit = true

                local bv = root:FindFirstChild("BangProtectBV")
                if not bv then
                    bv = Instance.new("BodyVelocity")
                    bv.Name = "BangProtectBV"
                    bv.Velocity = Vector3.zero
                    bv.MaxForce = Vector3.new(1e5, 0, 1e5)
                    bv.Parent = root
                end

                local speed = savedSpeed or 16
                local cf = WS.CurrentCamera.CFrame
                local fwd = Vector3.new(cf.LookVector.X, 0, cf.LookVector.Z)
                local rgt = Vector3.new(cf.RightVector.X, 0, cf.RightVector.Z)
                if fwd.Magnitude > 0 then fwd = fwd.Unit end
                if rgt.Magnitude > 0 then rgt = rgt.Unit end

                local vel = Vector3.zero
                if not UIS:GetFocusedTextBox() then
                    if UIS:IsKeyDown(Enum.KeyCode.W) then vel += fwd end
                    if UIS:IsKeyDown(Enum.KeyCode.S) then vel -= fwd end
                    if UIS:IsKeyDown(Enum.KeyCode.A) then vel -= rgt end
                    if UIS:IsKeyDown(Enum.KeyCode.D) then vel += rgt end
                end
                if UIS.TouchEnabled and hum.MoveDirection.Magnitude > 0 then
                    local md = hum.MoveDirection
                    vel += Vector3.new(md.X, 0, md.Z)
                end
                bv.Velocity = vel.Magnitude > 0 and vel.Unit * speed or Vector3.zero
            end)

            local tg = bTarget
            local tc = tg and tg.Character
            local myChar = LP.Character
            if myChar and tc then
                local myRoot = myChar:FindFirstChild("HumanoidRootPart")
                local targetRoot = tc:FindFirstChild("HumanoidRootPart")
                if myRoot and targetRoot then
                    myRoot.CFrame = targetRoot.CFrame * CFrame.new(0, 0, 1.5)
                end
            end
        end)

        bRespawnConn = LP.CharacterAdded:Connect(function(newChar)
            if not bActive then return end
            task.spawn(function()
                task.wait(0.5)
                if not bActive then return end
                bMakeAnim(newChar)
                task.wait(1.5)
                if not bActive then return end
                if not bAnimTrack or not bAnimTrack.IsPlaying then bMakeAnim() end
                task.wait(3.0)
                if not bActive then return end
                if not bAnimTrack or not bAnimTrack.IsPlaying then bMakeAnim() end
            end)
        end)

        return true
    end
    local function startFaceBang(t, speed)
        speed = speed or 8
        if not t then return end
        stopBang()
        local mh = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
        if not mh then return end
        bActive = true; bTarget = t; bTargetName = t.Name; bMode = "face"; bSavedCF = mh.CFrame
        pcall(function() WS.FallenPartsDestroyHeight = -math.huge end)
        bStopAnim()
        bThread = task.spawn(function()
            while bActive and bMode == "face" do
                local tg = bTarget
                if not tg or not tg.Parent or not tg.Character then
                    local r = bTargetName and getPlayer(bTargetName)
                    if r and r.Character then bTarget = r; tg = r else stopBang(); break end
                end
                local c = tg.Character
                local hrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
                local h = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
                if c and hrp and h and h.Health > 0 then
                    local head = c:FindFirstChild("Head")
                    if head then pcall(function()
                        h.Sit = true
                        local v = math.sin(tick() * speed)
                        hrp.CFrame = CFrame.new(head.Position + head.CFrame.LookVector * (1.2 + v * 1.25), head.Position)
                        hrp.Velocity = Vector3.zero
                    end) end
                end
                Rsvc.Heartbeat:Wait()
            end
        end)
    end

        Position = UDim2.fromOffset(10, 10),

    local nameLbl = label(card, "الاسم: ---", UDim2.fromOffset(6, 66), UDim2.new(1, -12, 0, 16), 12, C.text, Enum.Font.Gotham, Enum.TextXAlignment.Left)
    local uidLbl = label(card, "الايدي: ---", UDim2.fromOffset(6, 84), UDim2.new(1, -12, 0, 16), 11, C.dim, Enum.Font.Gotham, Enum.TextXAlignment.Left)

    make("UIPadding", {PaddingTop = UDim.new(0, 3), PaddingLeft = UDim.new(0, 3), PaddingRight = UDim.new(0, 3), PaddingBottom = UDim.new(0, 3)}, acList)

    local bangBtn = mkBtn(card, "BANG", UDim2.new(0.62, 6, 0, 66), UDim2.new(0.35, -12, 0, 30), C.green, C.white)
    bangBtn.TextSize = 13
    local faceBtn = mkBtn(card, "FACE BANG", UDim2.new(0.62, 6, 0, 102), UDim2.new(0.35, -12, 0, 30), C.green, C.white)

    local acButtons = {}

        acList.Visible = false

    local function showAC(matches, onPick)
        clearAC()
        if #matches == 0 then return end
        for i, m in ipairs(matches) do
            if i > 5 then break end
            local row = make("TextButton", {
                Size = UDim2.new(1, 0, 0, 26), BackgroundColor3 = C.panel2,
                Text = m.name .. (m.display ~= m.name and " (@" .. m.name .. ")" or ""),
                TextColor3 = C.text, Font = Enum.Font.GothamBold, TextSize = 11,
                TextXAlignment = Enum.TextXAlignment.Left, AutoButtonColor = false,
                LayoutOrder = i, ZIndex = 9,
            }, acList)
            corner(row, 4)
            row.MouseEnter:Connect(function() row.BackgroundColor3 = C.line end)
            row.MouseLeave:Connect(function() row.BackgroundColor3 = C.panel2 end)
            row.MouseButton1Click:Connect(function()
                onPick(m)
                clearAC()
            end)
            table.insert(acButtons, row)
        end
        local h = math.min(#matches, 5) * 28 + 6
        acList.Size = UDim2.new(1, -12, 0, h)
        acList.Visible = true
    end
    local function searchPlayers(q)
        local matches = {}
        local lq = q:lower()
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LP then
                local nm = p.Name:lower()
                local dn = (p.DisplayName or ""):lower()
                if nm:sub(1, #lq) == lq or (dn ~= "" and dn:sub(1, #lq) == lq) then
                    table.insert(matches, {name = p.Name, display = p.DisplayName, userId = p.UserId})
                end
            end
        end
        table.sort(matches, function(a, b) return a.name:lower() < b.name:lower() end)
        return matches
    end

        local txt = tBox.Text

                tBox.Text = m.name
                clearAC()
            end)
        else
            clearAC()
        end
        if #txt >= 1 then
            local now = tick()
            if now - lastLookup < 0.5 then return end
            lastLookup = now
            task.spawn(function()
                task.wait(0.4)
                if tBox.Text ~= txt then return end

                if ok and uid and tBox.Text == txt then
                    uidLbl.Text = "الايدي: " .. tostring(uid)
                    avatarImg.Image = string.format("rbxthumb://type=AvatarHeadShot&id=%d&w=150&h=150", uid)
                elseif tBox.Text == txt then
                    uidLbl.Text = "الايدي: غير موجود"
                    avatarImg.Image = ""
                end
            end)
        end
    end)

        else
            bangBtn.Text = "BANG"; bangBtn.BackgroundColor3 = C.green

        end
    end)

    local flyOn = false
    local flyBV, flyBG, flyConn
    local function toggleFly(on)
        if flyConn then flyConn:Disconnect(); flyConn = nil end
        if flyBV then flyBV:Destroy(); flyBV = nil end
        if flyBG then flyBG:Destroy(); flyBG = nil end
        flyOn = on
        local h = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
        if h then h.PlatformStand = false end
        if not on then return end
        local c = LP.Character; local r = c and c:FindFirstChild("HumanoidRootPart")
        if not (r and h) then return end
        flyBV = Instance.new("BodyVelocity"); flyBV.MaxForce = Vector3.new(1e9,1e9,1e9); flyBV.Velocity = Vector3.zero; flyBV.Parent = r
        flyBG = Instance.new("BodyGyro"); flyBG.MaxTorque = Vector3.new(1e9,1e9,1e9); flyBG.P = 10000; flyBG.Parent = r
        h.PlatformStand = true
        flyConn = Rsvc.Heartbeat:Connect(function()
            if not (flyBV and flyBG and LP.Character) then return end
            local cam = WS.CurrentCamera; flyBG.CFrame = cam.CFrame

        local function scanTool(c, lbl)
            if not c then return end
            for _, t in ipairs(c:GetChildren()) do
                if t:IsA("Tool") then
                    for _, d in ipairs(t:GetDescendants()) do
                        if d:IsA("RemoteEvent") and not seen[d] then
                            table.insert(remotes, {remote = d, tool = t, src = lbl .. ":" .. t.Name}); seen[d] = true
                        end
                    end
                end
            end
        end
        scanTool(LP.Character, "يد")
        scanTool(LP:FindFirstChildOfClass("Backpack"), "حقيبة")
        for _, d in ipairs(RS:GetDescendants()) do
            if d:IsA("RemoteEvent") and not seen[d] then
                local n = tostring(d.Name):lower()
                for _, k in ipairs(kws) do
                    if n:find(k,1,true) then table.insert(remotes, {remote=d, src="RS:" .. d.Name}); seen[d]=true; break end

    local antiCuffBtn = mkBtn(pages.protect, "🛡️ مضاد الكلبشة V2", UDim2.fromOffset(10, 85), UDim2.new(1, -20, 0, 40), C.white, C.black)
    antiCuffBtn.TextSize = 13

    local _cuff2Active     = false
    local _cuff2Conns      = {}
    local _savedCFrame     = nil
    local _resetInProgress = false
    local isTouchDevice    = UIS.TouchEnabled

    local function showNotification(msg)
        pcall(function()
            StarterGui:SetCore("SendNotification", {Title = "حماية كلبشة 2", Text = msg, Duration = 3})
        end)
    end

    local function isCuffed(char)
        for _, obj in ipairs(char:GetDescendants()) do
            if obj:IsA("WeldConstraint") then
                local p0, p1 = obj.Part0, obj.Part1
                if p0 and p0.Parent and not p0:IsDescendantOf(char) then return true end
                if p1 and p1.Parent and not p1:IsDescendantOf(char) then return true end
            end
        end

        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum and hum.PlatformStand then return true end

        local cv = char:FindFirstChild("Cuffing")
        if cv and cv:IsA("BoolValue") and cv.Value == true then return true end

        local anchorParts = {"UpperTorso", "LowerTorso", "HumanoidRootPart", "Torso"}
        for _, pName in ipairs(anchorParts) do
            local part = char:FindFirstChild(pName)
            if part and part:IsA("BasePart") and part.Anchored then
                return true
            end
        end
        return false
    end

    local function removeHDAdminScript(char)
        pcall(function()
            local s = char:FindFirstChild("HDAdminStarterCharacter")
            if s then
                s:Destroy()
                warn("[CuffProtect2] حُذف HDAdminStarterCharacter من الشخصية")
            end
        end)
    end

    local function doReset(hum)
        if _resetInProgress then return end
        _resetInProgress = true

        local savedCF = _savedCFrame

        showNotification("كلبشة مكتشفة! ريسيت تلقائي...")
        pcall(function() hum.Health = 0 end)

        task.spawn(function()
            local newChar = LP.CharacterAdded:Wait()
            local newRoot = newChar:WaitForChild("HumanoidRootPart", 8)
            local newHum  = newChar:WaitForChild("Humanoid", 8)

            if newRoot and newHum and savedCF then
                local t = tick()
                repeat task.wait(0.05) until newHum.Health > 0 or tick() - t > 6
                task.wait(0.2)

                pcall(function()
                    newRoot.CFrame = savedCF
                end)
                showNotification("رجعت لنفس المكان!")
            end

            task.wait(1.5)
            _resetInProgress = false
        end)
    end

    local function watchChar(char)
        if not char then return end
        local hum  = char:FindFirstChildOfClass("Humanoid")
        local root = char:FindFirstChild("HumanoidRootPart")
        if not hum or not root then return end

        removeHDAdminScript(char)

        local hdConn
        hdConn = char.ChildAdded:Connect(function(child)
            if child.Name == "HDAdminStarterCharacter" then
                task.defer(function()
                    pcall(function() child:Destroy() end)
                    warn("[CuffProtect2] HDAdminStarterCharacter اُضيف ثم حُذف فوراً")
                end)
            end
        end)
        table.insert(_cuff2Conns, hdConn)

        local anchoredParts = {"UpperTorso", "LowerTorso", "HumanoidRootPart", "Torso"}
        for _, pName in ipairs(anchoredParts) do
            local part = char:FindFirstChild(pName)
            if part and part:IsA("BasePart") then
                local anchorConn
                anchorConn = part:GetPropertyChangedSignal("Anchored"):Connect(function()
                    if not _cuff2Active or _resetInProgress then return end
                    if part.Anchored then
                        warn("[CuffProtect2] Anchored=true على: " .. pName)
                        pcall(function() anchorConn:Disconnect() end)
                        local curHum = char:FindFirstChildOfClass("Humanoid")
                        if curHum and curHum.Health > 0 then doReset(curHum) end
                    end
                end)
                table.insert(_cuff2Conns, anchorConn)
            end
        end

        _savedCFrame = root.CFrame

        task.spawn(function()
            while _cuff2Active and char.Parent do
                task.wait(2)
                if _resetInProgress then continue end
                local curRoot = char:FindFirstChild("HumanoidRootPart")
                local curHum  = char:FindFirstChildOfClass("Humanoid")
                if curRoot and curHum and curHum.Health > 0 then
                    local state = curHum:GetState()
                    if (state == Enum.HumanoidStateType.Running
                    or  state == Enum.HumanoidStateType.Landed
                    or  state == Enum.HumanoidStateType.Jumping
                    or  state == Enum.HumanoidStateType.Freefall)
                    and not isCuffed(char) then
                        _savedCFrame = curRoot.CFrame
                    end
                end
            end
        end)

        local monitor
        local wsConn = hum:GetPropertyChangedSignal("WalkSpeed"):Connect(function()
            if _resetInProgress or not _cuff2Active then return end
            if hum.WalkSpeed == 0 then
                task.delay(0.25, function()
                    if _resetInProgress or not _cuff2Active then return end
                    if hum.WalkSpeed == 0 then
                        if monitor then pcall(function() monitor:Disconnect() end) end
                        doReset(hum)
                    end
                end)
            end
        end)
        table.insert(_cuff2Conns, wsConn)

        local _lastCheck = 0
        monitor = Rsvc.Heartbeat:Connect(function()
            local now = tick()
            if now - _lastCheck < 0.5 then return end
            _lastCheck = now

            if not _cuff2Active or not char.Parent
            or hum.Health <= 0 or _resetInProgress then
                pcall(function() monitor:Disconnect() end)
                return
            end

            if isCuffed(char) then
                pcall(function() monitor:Disconnect() end)
                local curHum = char:FindFirstChildOfClass("Humanoid")
                if curHum and curHum.Health > 0 then doReset(curHum) end
            end
        end)

        table.insert(_cuff2Conns, monitor)

        local descConn
        descConn = char.DescendantAdded:Connect(function(obj)
            if _resetInProgress or not _cuff2Active then
                pcall(function() descConn:Disconnect() end)
                return
            end
            if obj:IsA("WeldConstraint") then
                task.defer(function()
                    if _resetInProgress then return end
                    local p0, p1 = obj.Part0, obj.Part1
                    if (p0 and p0.Parent and not p0:IsDescendantOf(char))
                    or (p1 and p1.Parent and not p1:IsDescendantOf(char)) then
                        pcall(function() descConn:Disconnect() end)
                        pcall(function() monitor:Disconnect() end)
                        local curHum = char:FindFirstChildOfClass("Humanoid")
                        if curHum and curHum.Health > 0 then doReset(curHum) end
                    end
                end)
            end
        end)
        table.insert(_cuff2Conns, descConn)

        local wsConn2
        wsConn2 = WS.DescendantAdded:Connect(function(obj)
            if _resetInProgress or not _cuff2Active then return end
            if not obj:IsA("WeldConstraint") then return end
            task.defer(function()
                if _resetInProgress then return end
                local p0, p1 = obj.Part0, obj.Part1

                local charInvolved = (p0 and p0:IsDescendantOf(char))
                                  or (p1 and p1:IsDescendantOf(char))
                if not charInvolved then return end

                local externalFound = (p0 and p0.Parent and not p0:IsDescendantOf(char))
                                   or (p1 and p1.Parent and not p1:IsDescendantOf(char))
                if externalFound then
                    warn("[CuffProtect2] WeldConstraint خارجي مكتشف في Workspace!")
                    pcall(function() wsConn2:Disconnect() end)
                    pcall(function() monitor:Disconnect() end)
                    local curHum = char:FindFirstChildOfClass("Humanoid")
                    if curHum and curHum.Health > 0 then doReset(curHum) end
                end
            end)
        end)
        table.insert(_cuff2Conns, wsConn2)

        for _, bodyPart in ipairs(char:GetChildren()) do
            if bodyPart:IsA("BasePart") then
                local partConn
                partConn = bodyPart.DescendantAdded:Connect(function(obj)
                    if _resetInProgress or not _cuff2Active then
                        pcall(function() partConn:Disconnect() end)
                        return
                    end
                    if obj:IsA("WeldConstraint") then
                        task.defer(function()
                            if _resetInProgress then return end
                            local p0, p1 = obj.Part0, obj.Part1
                            if (p0 and not p0:IsDescendantOf(char))
                            or (p1 and not p1:IsDescendantOf(char)) then
                                warn("[CuffProtect2] Weld في جزء: " .. bodyPart.Name)
                                pcall(function() partConn:Disconnect() end)
                                pcall(function() monitor:Disconnect() end)
                                local curHum = char:FindFirstChildOfClass("Humanoid")
                                if curHum and curHum.Health > 0 then doReset(curHum) end
                            end
                        end)
                    end
                end)
                table.insert(_cuff2Conns, partConn)
            end
        end

        do
            local _fb_lastSafeCF      = root.CFrame
            local _fb_lastPos         = root.Position
            local _fb_lastWalkSpeed   = hum.WalkSpeed
            local _fb_lastJumpPower   = hum.JumpPower
            local _fb_lastState       = hum:GetState()
            local _fb_cooldownUntil   = 0
            local _fb_COOLDOWN        = 6
            local _fb_frozenTick      = nil
            local _fb_FREEZE_CONFIRM  = 1.2
            local _fb_resetDone       = false

            local _fb_score           = 0

            local function _fb_doReset(reason)
                if _resetInProgress  then return end
                if _fb_resetDone     then return end
                if tick() < _fb_cooldownUntil then return end

                if not _cuff2Active then return end

                _fb_resetDone     = true
                _resetInProgress  = true
                _fb_cooldownUntil = tick() + _fb_COOLDOWN

                local savedCF = _savedCFrame

                warn("[CuffProtect2-Fallback] تفعّل بسبب: " .. tostring(reason))
                showNotification("[Fallback] تجميد مكتشف: " .. reason)

                pcall(function() hum.Health = 0 end)

                task.spawn(function()
                    local newChar = LP.CharacterAdded:Wait()
                    local newRoot = newChar:WaitForChild("HumanoidRootPart", 8)
                    local newHum  = newChar:WaitForChild("Humanoid", 8)

                    if newRoot and newHum and savedCF then
                        local t = tick()
                        repeat task.wait(0.05) until newHum.Health > 0 or tick()-t > 6
                        task.wait(0.2)
                        pcall(function() newRoot.CFrame = savedCF end)
                        showNotification("[Fallback] رجعت للموقع الآمن!")
                        warn("[CuffProtect2-Fallback] تم الارجاع للموقع: " .. tostring(savedCF))
                    end

                    task.wait(1.5)
                    _resetInProgress = false
                    _fb_resetDone    = false
                    _fb_score        = 0
                    _fb_frozenTick   = nil
                end)
            end

            local NORMAL_STATES = {
                [Enum.HumanoidStateType.Running]          = true,
                [Enum.HumanoidStateType.Landed]           = true,
                [Enum.HumanoidStateType.Jumping]          = true,
                [Enum.HumanoidStateType.Freefall]         = true,
                [Enum.HumanoidStateType.Climbing]         = true,
                [Enum.HumanoidStateType.Swimming]         = true,
                [Enum.HumanoidStateType.RunningNoPhysics] = true,
            }

            local SUSPECT_STATES = {
                [Enum.HumanoidStateType.PlatformStanding]  = true,
                [Enum.HumanoidStateType.Seated]            = true,
                [Enum.HumanoidStateType.Physics]           = true,
                [Enum.HumanoidStateType.StrafingNoPhysics] = true,
            }

            local _fb_lastTick = 0
            local _fb_monitor
            _fb_monitor = Rsvc.Heartbeat:Connect(function()
                local now = tick()
                if now - _fb_lastTick < 0.3 then return end
                _fb_lastTick = now

                if not _cuff2Active
                or not char.Parent
                or hum.Health <= 0
                or _resetInProgress
                or _fb_resetDone then
                    pcall(function() _fb_monitor:Disconnect() end)
                    return
                end

                pcall(function()
                    local curRoot = char:FindFirstChild("HumanoidRootPart")
                    local curHum  = char:FindFirstChildOfClass("Humanoid")
                    if not curRoot or not curHum or curHum.Health <= 0 then
                        pcall(function() _fb_monitor:Disconnect() end)
                        return
                    end

                    local curState = curHum:GetState()
                    local curPos   = curRoot.Position
                    local posDiff  = (curPos - _fb_lastPos).Magnitude

                    if NORMAL_STATES[curState] and not isCuffed(char) then
                        _savedCFrame    = curRoot.CFrame
                        _fb_lastSafeCF  = curRoot.CFrame
                        _fb_score       = 0
                        _fb_frozenTick  = nil
                    end

                    local reasonParts = {}
                    local score = 0

                    if curHum.PlatformStand == true then
                        score += 3
                        table.insert(reasonParts, "PlatformStand=true")
                    end

                    if SUSPECT_STATES[curState] then
                        score += 2
                        table.insert(reasonParts, "State="..tostring(curState))
                    end

                    if curHum.WalkSpeed == 0 and _fb_lastWalkSpeed > 0 then
                        score += 2
                        table.insert(reasonParts, "WalkSpeed→0")
                    end

                    if curHum.JumpPower == 0 and _fb_lastJumpPower > 0 then
                        score += 1
                        table.insert(reasonParts, "JumpPower→0")
                    end

                    if not isTouchDevice then
                        local tryMove = UIS:IsKeyDown(Enum.KeyCode.W)
                                     or UIS:IsKeyDown(Enum.KeyCode.A)
                                     or UIS:IsKeyDown(Enum.KeyCode.S)
                                     or UIS:IsKeyDown(Enum.KeyCode.D)
                        if tryMove and posDiff < 0.03 and curHum.WalkSpeed > 0 then
                            score += 2
                            table.insert(reasonParts, "FrozenWhileMoving")
                        end
                    end

                    if curRoot.Parent ~= char then
                        score += 4
                        table.insert(reasonParts, "RootReparented")
                    end

                    for _, obj in ipairs(curRoot:GetChildren()) do
                        if obj:IsA("AlignPosition")
                        or obj:IsA("AlignOrientation")
                        or obj:IsA("VectorForce")
                        or obj:IsA("BodyPosition")
                        or obj:IsA("BodyGyro") then
                            score += 2
                            table.insert(reasonParts, "Constraint:"..obj.ClassName)
                            break
                        end
                    end

                    local vel = curRoot.AssemblyLinearVelocity
                    if vel.Magnitude < 0.1 and posDiff < 0.03 and curHum.WalkSpeed > 0 then
                        score += 1
                        table.insert(reasonParts, "ZeroVelocity")
                    end

                    local anchorCheck = {"UpperTorso","LowerTorso","HumanoidRootPart","Torso"}
                    for _, pn in ipairs(anchorCheck) do
                        local p = char:FindFirstChild(pn)
                        if p and p:IsA("BasePart") and p.Anchored then
                            score += 5
                            table.insert(reasonParts, "Anchored:"..pn)
                            break
                        end
                    end

                    _fb_lastPos       = curPos
                    _fb_lastWalkSpeed = curHum.WalkSpeed
                    _fb_lastJumpPower = curHum.JumpPower
                    _fb_lastState     = curState

                    if score >= 4 then
                        if _fb_frozenTick == nil then
                            _fb_frozenTick = tick()
                        elseif tick() - _fb_frozenTick >= _fb_FREEZE_CONFIRM then
                            local reason = table.concat(reasonParts, " + ")
                            pcall(function() _fb_monitor:Disconnect() end)
                            _fb_doReset(reason)
                        end
                    else
                        _fb_frozenTick = nil
                        _fb_score      = 0
                    end
                end)
            end)

            table.insert(_cuff2Conns, _fb_monitor)

            local _fb_propConn
            _fb_propConn = hum:GetPropertyChangedSignal("PlatformStand"):Connect(function()
                if not _cuff2Active or _resetInProgress or _fb_resetDone then return end
                if hum.PlatformStand == true then
                    warn("[CuffProtect2-Fallback] PropertyChanged: PlatformStand → true")
                    pcall(function() _fb_propConn:Disconnect() end)
                    _fb_doReset("PlatformStand_PropertyChanged")
                end
            end)
            table.insert(_cuff2Conns, _fb_propConn)

            local _fb_wsConn
            _fb_wsConn = hum:GetPropertyChangedSignal("WalkSpeed"):Connect(function()
                if not _cuff2Active or _resetInProgress or _fb_resetDone then return end

                if hum.WalkSpeed == 0 then
                    task.delay(0.3, function()
                        if not _cuff2Active or _resetInProgress or _fb_resetDone then return end
                        if hum.WalkSpeed == 0 and
                           (hum.PlatformStand or SUSPECT_STATES[hum:GetState()]) then
                            warn("[CuffProtect2-Fallback] WalkSpeed→0 مع حالة مشبوهة")
                            pcall(function() _fb_wsConn:Disconnect() end)
                            _fb_doReset("WalkSpeed=0+SuspectState")
                        end
                    end)
                end
            end)
            table.insert(_cuff2Conns, _fb_wsConn)
        end
    end

    antiCuffBtn.MouseButton1Click:Connect(function()
        _cuff2Active = not _cuff2Active

        if _cuff2Active then
            antiCuffBtn.Text = "⛔ إيقاف مضاد الكلبشة"
            antiCuffBtn.BackgroundColor3 = C.green
            antiCuffBtn.TextColor3 = C.white
            showNotification("حماية كلبشه2 مفعلة")

            watchChar(LP.Character)

            table.insert(_cuff2Conns, LP.CharacterAdded:Connect(function(newChar)
                if _cuff2Active then

        startBtn.MouseButton1Click:Connect(function()
            if isRunning then return end
            local cmd = cmdIn.Text
            if cmd == "" then statusLbl.Text = "الحالة: اكتب أمر!"; return end
            local speed = math.clamp(tonumber(speedIn.Text) or 0.5, 0.1, 5)
            isRunning = true; token = token + 1
            local myToken = token
            statusLbl.Text = "الحالة: يعمل..."; statusLbl.TextColor3 = C.white
            startBtn.BackgroundColor3 = C.panel; startBtn.TextColor3 = C.dim
            task.spawn(function()
                local count = 0
                while isRunning and token == myToken do
                    pcall(sendCmdRaw, cmd)
                    count = count + 1
                    statusLbl.Text = "الحالة: يعمل (#" .. count .. ")"
                    task.wait(speed)
                end
            end)
        end)
        stopBtn.MouseButton1Click:Connect(function()
            isRunning = false; token = token + 1
            statusLbl.Text = "الحالة: متوقف"; statusLbl.TextColor3 = C.dim
            startBtn.BackgroundColor3 = C.white; startBtn.TextColor3 = C.black
        end)
    end

    local isSpamming = false
    local spamToken = 0
    local spamBtn = mkBtn(sendFrame, "بدء الإرسال", UDim2.fromOffset(0, 254), UDim2.new(1, -20, 0, 40), C.panel2, C.text)
    spamBtn.TextSize = 14
    spamBtn.MouseButton1Click:Connect(function()
        if isSpamming then
            isSpamming = false; spamToken = spamToken + 1
            spamBtn.Text = "بدء الإرسال"; spamBtn.BackgroundColor3 = C.panel2; spamBtn.TextColor3 = C.text
        else
            local msg = cmdInpBox.Text
            if msg == "" then return end
            isSpamming = true; spamToken = spamToken + 1
            local myToken = spamToken
            spamBtn.Text = "إيقاف الإرسال"; spamBtn.BackgroundColor3 = C.white; spamBtn.TextColor3 = C.black
            task.spawn(function() while isSpamming and spamToken == myToken do pcall(sendCmdRaw, msg); task.wait(0.05) end end)
        end
    end)

    local addRow = make("Frame", {Size = UDim2.new(1, 0, 0, 30), Position = UDim2.fromOffset(0, 0), BackgroundTransparency = 1}, buildFrame)

    local cmdListScroll = make("ScrollingFrame", {Position = UDim2.fromOffset(0, 40), Size = UDim2.new(1, 0, 1, -40), BackgroundColor3 = C.panel2, BorderSizePixel = 0, ScrollBarThickness = 3, ScrollBarImageColor3 = C.text, CanvasSize = UDim2.new(0, 0, 0, 0), AutomaticCanvasSize = Enum.AutomaticSize.Y}, buildFrame)

    make("UIListLayout", {Padding = UDim.new(0, 4), SortOrder = Enum.SortOrder.LayoutOrder}, cmdListScroll)
    make("UIPadding", {PaddingTop = UDim.new(0, 5), PaddingLeft = UDim.new(0, 5), PaddingRight = UDim.new(0, 5), PaddingBottom = UDim.new(0, 5)}, cmdListScroll)
    local function refreshList()
        for _, c in ipairs(cmdListScroll:GetChildren()) do if c:IsA("Frame") then c:Destroy() end end
        for i, cmd in ipairs(spamCommands) do
            local row = make("Frame", {Size = UDim2.new(1, 0, 0, 35), BackgroundColor3 = C.bg, LayoutOrder = i, ZIndex = 6}, cmdListScroll)
            corner(row, 6)
            label(row, cmd.text .. " x" .. cmd.repeatCount, UDim2.fromOffset(10, 8), UDim2.new(0.5, 0, 0, 20), 11, C.text, Enum.Font.Code, Enum.TextXAlignment.Left)
            local delBtn = make("TextButton", {Size = UDim2.new(0, 50, 0, 24), Position = UDim2.new(1, -60, 0.5, -12), BackgroundColor3 = C.panel2, Text = "حذف", TextColor3 = C.text, Font = Enum.Font.GothamBold, TextSize = 10, ZIndex = 7}, row)
            corner(delBtn, 4)
            delBtn.MouseButton1Click:Connect(function() table.remove(spamCommands, i); saveSpamConfig(); refreshList() end)
        end
    end
    genCmdsBtn.MouseButton1Click:Connect(function()
        local tgt = getSelectedTargets()
        if tgt == "" then tgt = "TARGET" end
        local generated = {}
        for _, cmd in ipairs(spamCommands) do
            local actual
            if cmd.template and cmd.template ~= "" then actual = string.gsub(cmd.template, "{target}", tgt)
            else actual = cmd.text .. " " .. tgt end
            for _ = 1, (cmd.repeatCount or 1) do table.insert(generated, actual) end
        end
        cmdInpBox.Text = table.concat(generated, " ")
    end)
    addCmdBtn.MouseButton1Click:Connect(function()
        local t = newCmdInp.Text
        if t == "" then return end
        table.insert(spamCommands, {text = t, repeatCount = newCmdRepeat, template = ""})
        saveSpamConfig(); refreshList(); newCmdInp.Text = ""
    end)
    refreshList()
